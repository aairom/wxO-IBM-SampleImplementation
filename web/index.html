For developers

Embedding the chat widget

Add this snippet to any HTML page to embed the IBM Bob Assistant chat. The values are generated by running bash scripts/get_webchat_embed.sh. 

// 1. Configure the widget
window.wxOConfiguration = {
  orchestrationID: "<from: orchestrate channels webchat embed>",
  hostURL:         "https://api.eu-de.watson-orchestrate.cloud.ibm.com",
  rootElementID:   "wxo-chat-root",
  showLauncher:    true,
  crn:             "crn:v1:bluemix:public:...",
  deploymentPlatform: "ibmcloud",
  chatOptions: {
    agentId:            "<agent UUID>",
    agentEnvironmentId: "<live env UUID>"
  }
};

// 2. Load the wxO chat loader asynchronously
setTimeout(function() {
  const script = document.createElement("script");
  script.src = `${window.wxOConfiguration.hostURL}/wxochat/wxoLoader.js?embed=true`;
  script.addEventListener("load", () => wxoLoader.init());
  document.head.appendChild(script);
}, 0);
